1.Verilog Code


module alu_4bit(
    input [3:0] A,
    input [3:0] B,
    input [3:0] sel,
    output reg [3:0] result,
    output reg carry
);

reg [4:0] temp;

always @(*) begin

    carry = 0;
    temp = 0;

    case(sel)

        4'b0000: temp = A + B;
        4'b0001: temp = A - B;

        4'b0010: temp = {1'b0, A & B};
        4'b0011: temp = {1'b0, A | B};
        4'b0100: temp = {1'b0, A ^ B};

        4'b0101: temp = {1'b0, ~(A & B)};
        4'b0110: temp = {1'b0, ~(A | B)};
        4'b0111: temp = {1'b0, ~(A ^ B)};

        4'b1000: temp = {1'b0, A};
        4'b1001: temp = {1'b0, ~A};

        4'b1010: temp = A + 1'b1;
        4'b1011: temp = A - 1'b1;

        4'b1100: temp = {1'b0, A << 1};
        4'b1101: temp = {1'b0, A >> 1};

        4'b1110: temp = B + 1'b1;
        4'b1111: temp = {1'b0, B >> 1};

        default: temp = 5'b00000;

    endcase

    result = temp[3:0];
    carry = temp[4];

end

endmodule

module alu_4bit_tb;

reg [3:0] B;

    .A(A),
    .B(B),
    .sel(sel),
    .result(result),
    .carry(carry)

    $monitor("A=%b B=%b sel=%b result=%b carry=%b",
