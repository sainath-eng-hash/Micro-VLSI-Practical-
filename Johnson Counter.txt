1.Verilog Code


module johnson_counter(
    input clk,
    input reset,
    output reg [3:0] Q
);

always @(posedge clk)
begin
    if(reset)
        Q <= 4'b0000;
    else
        Q <= {Q[2:0], ~Q[3]};
end

endmodule





2.Test Bench 



module johnson_counter_tb;

reg clk, reset;
wire [3:0] Q;

johnson_counter DUT(
    .clk(clk),
    .reset(reset),
    .Q(Q)
);

always #5 clk = ~clk;

initial begin

    clk = 0;
    reset = 1;

    #10 reset = 0;

    #100;

    $stop;

end

initial begin
    $monitor("Time=%0t Q=%b", $time, Q);
end

endmodule
