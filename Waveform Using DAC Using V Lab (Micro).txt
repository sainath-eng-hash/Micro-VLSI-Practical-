1.Sawtooth Waveform Using DAC



#include <reg51.h>

#define DAC P1

void delay(void)
{
    unsigned int i;

    for(i = 0; i < 500; i++);
}

void main(void)
{
    unsigned char i;

    while(1)
    {
        for(i = 0; i < 255; i++)
        {
            DAC = i;
            delay();
        }
    }
}




2.Sine waveform Using DAC




#include <reg51.h>

#define DAC P1

unsigned char sine_table[] =
{
    128, 160, 192, 224,
    244, 255, 244, 224,
    192, 160, 128, 96,
    64, 32, 12, 0
};

void delay(void)
{
    unsigned int i;

    for(i = 0; i < 500; i++);
}

void main(void)
{
    unsigned char i;

    while(1)
    {
        for(i = 0; i < 16; i++)
        {
            DAC = sine_table[i];
            delay();
        }
    }
}
